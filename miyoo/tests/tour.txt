# Screen tour for checking the Miyoo layouts: opens each screen and holds it so the
# screenshot hook (ES_SHOT_EVERY) catches it. Copy into a config's plugins folder
# next to the integration tests, then run one of the tests below with --debug.

test "Tour Hold"
	status partial
	sequence
		apply
			"tour: wait" = 10
		label "hold"
		input
		apply
			"tour: wait" = "tour: wait" - 1
		branch "hold"
			"tour: wait" > 0
